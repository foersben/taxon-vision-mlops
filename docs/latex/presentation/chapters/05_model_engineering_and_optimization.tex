\begin{frame}{Model Engineering}{Linear Probing vs Full Fine-Tuning Tradeoff}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{Full Fine-Tuning}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Compute:} Requires 4-hour GPU jobs to update $\sim$86M parameters.
                    \item \textbf{Hardware:} Consumes up to 16 GB VRAM.
                    \item \textbf{Risk:} Catastrophic forgetting of generalized representations.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Linear Probing}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Compute:} Trains in $\sim$10-second runs on pre-cached embeddings.
                    \item \textbf{Hardware:} Requires $<$1 GB VRAM (freezes backbone).
                    \item \textbf{Optimization:} High-throughput hyperparameter search.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.4cm}
    \begin{block}{Engineering Rationale}
        \footnotesize
        Pre-cached embeddings combined with a convex classification head drastically reduce operational latency, unlocking Bayesian hyperparameter sweeps that are impossible under full fine-tuning constraints.
    \end{block}

    \script{Let's compare the economics of linear probing versus full fine-tuning. Full fine-tuning requires backpropagating through 86 million transformer parameters, consuming 14 gigabytes of VRAM and 4 hours of GPU time per run. Linear probing trains a single linear projection matrix on pre-extracted embeddings. It uses 500 megabytes of VRAM and finishes in 8 seconds. This 1,000x speedup allows us to explore hyperparameter spaces thoroughly rather than settling for un-tuned defaults.}
    \qa{Can linear probing adapt if the test domain has different lighting or camera sensors?}{Because the foundation model was pre-trained on diverse web-scale imagery, its internal representations already generalize well across lighting and sensor variations without needing backbone parameter updates.}
\end{frame}

\begin{frame}{Backbone Factory}{Vision Backbone Registry: Configuration \& Parameters}
    \vspace{-0.4cm}
    \begin{block}{Config-Driven Architecture}

        A unified factory instantiates any backbone with a frozen trunk ($\nabla_\theta = 0$), yielding dense L2-normalized feature representations.

    \vspace{-0.4cm}
    \begin{block}{The 6 Canonical Models (strictly matching \texttt{config/default\_config.yaml})}
        \tiny
        \begin{itemize}

            \item \texttt{dinov3} (\texttt{vit\_small\_patch16\_dinov3}): $D=384$, $224 \times 224$, Register tokens (reg4), dense structural representations.
            \item \texttt{dinov3\_vits14} (\texttt{vit\_small\_patch16\_dinov3}): $D=384$, $224 \times 224$, 16x16 patch ViT-S variant.
            \item \texttt{dinov2\_vits14} (\texttt{vit\_small\_patch14\_dinov2.lvd142m}): $D=384$, $224 \times 224$, 14x14 patches, LVD-142M self-supervised.
            \item \texttt{mobilenetv4\_conv\_small} (\texttt{mobilenetv4\_conv\_small.e2400\_r224\_in1k}): $D=960$, $224 \times 224$, Universal Inverted Bottleneck (UIB) + Global Average Pooling (GAP), edge/CPU optimized.
            \item \texttt{efficientnet\_b0} (\texttt{efficientnet\_b0.ra\_in1k}): $D=1280$, $224 \times 224$, Depthwise Separable Conv + Squeeze-and-Excitation, classic compound scaling.
        \end{itemize}
    \end{block}

    \vspace{-0.4cm}
    \begin{block}{Distinct vs. Shared Parameters}
        \tiny
        \begin{itemize}
            \item \emph{Distinct:} Latent dimension $D$ (384 to 1280), feature pooling (CLS token vs. Register tokens vs. GAP), normalization statistics.
            \item \emph{Shared:} Frozen trunk ($\nabla_\theta = 0$), standardized input tensor ($3 \times 224 \times 224$), unit L2 normalization, convex linear classification interface.
        \end{itemize}
    \end{block}

    \vspace{-0.3cm}
    \statusdone
    \script{We manage six canonical vision backbones through a unified factory. Each backbone is frozen, sharing a standard 224 by 224 input and outputting an L2-normalized feature vector. However, they vary significantly in their latent dimensions—from 384 in DINOv3 to 1280 in EfficientNet-B0—and pooling mechanisms, giving us a spectrum of options from heavy biological pre-training like BioCLIP-2 to edge-optimized MobileNetV4.}
    \qa{Why do the DINO models use register tokens instead of a standard CLS token?}{Register tokens allow the vision transformer to store global image information and discard uninformative background patches without corrupting the semantic representation, which is common when using a single CLS token.}
    \qa{How does freezing the trunk ($\nabla_\theta = 0$) impact our ability to deploy to new edge devices?}{A frozen trunk allows us to compile the feature extraction layer to highly optimized ONNX graphs once, then swap out only the lightweight linear classification head for different target applications.}
\end{frame}

\begin{frame}{Model Architecture}{Architectural Showdown: Vision Transformers vs. Modern CNNs}
    \vspace{-0.2cm}
    \begin{block}{Visual \& Structural Contrast}
        \scriptsize
        Patch tokenization with multi-head self-attention vs. Hierarchical inductive convolutional kernels.
    \end{block}

    \vspace{-0.2cm}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{Vision Transformers (ViTs)}
                \scriptsize
                \textbf{Models:} BioCLIP-2, DINOv3, DINOv2\\[0.1cm]
                \textbf{Strengths:}
                \begin{itemize}
                    \item Global receptive field from layer 1.
                    \item Rich web-scale morphological priors.
                    \item Register tokens prevent feature collapse.
                \end{itemize}
                \vspace{0.1cm}
                \textbf{Costs:}
                \begin{itemize}
                    \item Quadratic attention memory scaling.
                    \item Higher peak VRAM.
                    \item Lack of native inductive translation bias on small datasets.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Modern CNNs}
                \scriptsize
                \textbf{Models:} MobileNetV4, EfficientNet-B0\\[0.1cm]
                \textbf{Strengths:}
                \begin{itemize}
                    \item Native shift/translation invariance.
                    \item Linear memory scaling with image size.
                    \item Optimal for edge CPU deployment and low-power inference.
                \end{itemize}
                \vspace{0.1cm}
                \textbf{Costs:}
                \begin{itemize}
                    \item Fixed local receptive fields.
                    \item Requires deeper networks to capture long-range contextual relationships.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{-0.2cm}
    \statusdone
    \script{When selecting an architecture, we face a fundamental tradeoff. Vision Transformers, like BioCLIP-2 and DINO, offer a global receptive field from the very first layer, capturing long-range dependencies across the image, but suffer from quadratic memory scaling due to self-attention. Conversely, modern CNNs like MobileNetV4 provide native translation invariance and scale linearly, making them ideal for edge CPU deployment, though they require much deeper networks to understand global context.}
    \qa{If ViTs lack translation invariance, how do they learn to recognize a bird regardless of its position in the image?}{ViTs compensate for the lack of native translation invariance through massive scale pre-training, learning the necessary spatial relationships directly from the web-scale data rather than relying on structural priors.}
    \qa{Why does MobileNetV4 scale linearly with image size while ViTs scale quadratically?}{CNN memory scales linearly because convolutional filters operate on local neighborhoods. ViT memory scales quadratically because the self-attention mechanism computes the relationship between every patch and every other patch.}
\end{frame}

\begin{frame}{Loss Formulation}{Class-Balanced Loss \& Effective Number of Samples}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Mathematical Formulation}
                \footnotesize
                \begin{itemize}
                    \item Standard cross-entropy collapses under long-tail imbalance.
                    \item Naïve inverse frequency fails due to redundancy in head classes.
                    \item \textbf{Effective Samples:} $E_n = \frac{1 - \beta^n}{1 - \beta}$
                    \item \textbf{Loss Weight:} $W_n \propto \frac{1}{E_n} = \frac{1 - \beta}{1 - \beta^n}$
                \end{itemize}
            \end{block}
            \vspace{-0.2cm}
            \begin{block}{Engineering Rationale}
                \footnotesize
                This prevents gradient starvation for rare classes while bounding the penalty for hyper-abundant head classes.
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \vspace{-0.2cm}
            \begin{figure}
                \centering
                \begin{tikzpicture}
                    \begin{axis}[
                        width=\textwidth,
                        height=5.0cm,
                        xmode=log,
                        xlabel={Class Frequency $n$ (Log)},
                        ylabel={Weight $W_n$},
                        domain=1:10000,
                        samples=100,
                        grid=both,
                        title={Class-Balanced Loss Weights},
                        title style={font=\footnotesize\bfseries},
                        label style={font=\footnotesize},
                        tick label style={font=\scriptsize},
                        legend style={font=\scriptsize, at={(0.95, 0.85)}, anchor=north east, fill=white, fill opacity=0.8, text opacity=1}
                    ]
                    \addplot[thick, color=deepblue] {(1 - 0.99)/(1 - 0.99^x)};
                    \addlegendentry{$\beta = 0.99$}
                    \addplot[thick, color=signalorange] {(1 - 0.999)/(1 - 0.999^x)};
                    \addlegendentry{$\beta = 0.999$}
                    \end{axis}
                \end{tikzpicture}
            \end{figure}
        \end{column}
    \end{columns}

    \script{Standard cross-entropy loss fails on long-tail data because the gradients from 10,000 common robins overwhelm the gradients from 3 rare warblers. But if you simply weight by inverse frequency, you over-weight outliers and make training unstable. We use Class-Balanced Loss, which calculates the 'effective number of samples' using a hyperparameter beta. It mathematically accounts for information redundancy in large classes, providing stable, well-balanced gradients.}
    \qa{What is the typical value of $\beta$ in our pipelines?}{We tune $\beta$ via Bayesian optimization in the range $[0.99, 0.9999]$. When $\beta = 0$, it degrades to unweighted loss; as $\beta \to 1$, it approaches inverse frequency.}
\end{frame}

\begin{frame}{Optimization}{Optimization Dynamics \& Best-Weight Restoration}
    \begin{block}{Optimization Strategy}
        \footnotesize
        \begin{itemize}
            \item \textbf{AdamW Optimizer:} Incorporates decoupled weight decay for robust regularization.
            \item \textbf{Learning Rate Schedule:} Linear warmup prevents early gradient divergence; cosine annealing refines convergence in later epochs.
            \item \textbf{Early Stopping:} Continuously monitors validation loss over a patience window to prevent overfitting.
        \end{itemize}
    \end{block}

    \begin{block}{Deterministic Best-Weight Restoration}
        \footnotesize
        \begin{itemize}
            \item \textbf{Mechanism:} The runner automatically caches intermediate checkpoints and restores the exact weights corresponding to the minimum validation loss.
            \item \textbf{Engineering Rationale:} The final epoch is rarely the optimal one. Best-weight restoration strictly guarantees that the deployed checkpoint maximizes generalization on holdout data.
        \end{itemize}
    \end{block}

    \script{Even with pre-cached embeddings, optimization dynamics matter. If you use a high learning rate on epoch 1, you can destabilize the classifier head weights. Our runner implements linear warmup to gently ramp up learning rates, followed by cosine annealing. When validation loss plateaus, our early stopping mechanism triggers and automatically restores the best-performing historical weights, guaranteeing that our deployed checkpoint is optimal.}
    \qa{Why not simply save weights at the very end of training?}{The final epoch is frequently overfitted to the training data. Best-weight restoration guarantees that we retain the checkpoint with the highest generalization performance on holdout data.}
\end{frame}

\begin{frame}{AutoML \& Tuning}{Bayesian Optimization: Optuna TPE with ASHA Pruning}
    \begin{block}{Bayesian Hyperparameter Search}
        \footnotesize
        \begin{itemize}
            \item \textbf{TPE Algorithm:} Optuna uses the Tree-structured Parzen Estimator (TPE) to model probability distributions of promising regions (learning rate, $\beta$, dropout, weight decay).
            \item \textbf{Efficiency:} Vastly outperforms brute-force grid search (exponential complexity) or unguided random search.
        \end{itemize}
    \end{block}

    \begin{block}{ASHA Pruning Mechanism}
        \footnotesize
        \begin{itemize}
            \item \textbf{Operation:} The Asynchronous Successive Halving Algorithm (ASHA) actively monitors intermediate epoch performance.
            \item \textbf{Engineering Rationale:} Aggressively terminates underperforming trials early, reallocating GPU cycles to promising candidates and maximizing search efficiency.
        \end{itemize}
    \end{block}

    \script{Junior practitioners often tune learning rates by hand, guessing numbers in a script. That is unscientific and wastes compute. We integrate Optuna with Bayesian Tree-structured Parzen Estimators. Optuna builds a probabilistic model of which hyperparameters produce the highest validation scores. Even better, our ASHA pruner monitors intermediate epochs: if a trial is performing poorly by epoch 3, it kills it immediately and starts the next candidate, multiplying our search efficiency.}
    \qa{How many trials do we typically execute during an optimization run?}{Because our linear probe trains in seconds, we can run 50 to 100 trials in under 10 minutes, thoroughly exploring the hyperparameter surface.}
\end{frame}

\begin{frame}{Evaluation Metrics}{Macro PR-AUC vs Single-Threshold F1 Score}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Metric Limitations}
                \footnotesize
                \begin{itemize}
                    \item \textbf{F1 Score:} Brittle because it relies on a static threshold (e.g., 0.5), severely penalizing rare classes whose optimal threshold is much lower.
                    \item \textbf{ROC-AUC:} Misleading in extreme imbalance; true negatives overwhelm the metric, producing falsely optimistic scores.
                \end{itemize}
            \end{block}
            \vspace{-0.2cm}
            \begin{block}{Macro PR-AUC Advantage}
                \footnotesize
                \begin{itemize}
                    \item Evaluates the area under the Precision-Recall curve across \emph{all} recall thresholds independently.
                    \item \textbf{Engineering Rationale:} Provides a robust, threshold-invariant promotion gate reflecting positive performance.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \vspace{-0.2cm}
            \begin{figure}
                \centering
                \begin{tikzpicture}
                    \begin{axis}[
                        width=\textwidth,

        \end{column}

    \script{A crucial question for any data scientist: how do you decide whether Model B is better than Model A? Many people look at accuracy or F1-score. But F1 requires choosing an arbitrary threshold, like 0.5. For a rare species with only three images, the optimal decision threshold might be 0.12! If you evaluate with a 0.5 threshold, F1 drops to zero. Macro PR-AUC measures the entire precision-recall curve across all possible thresholds for every single class, giving us a true, un-biased measure of classification power.}
    \qa{Why not use ROC-AUC instead of PR-AUC?}{In extreme imbalance with 10,000 classes, each class has a massive pool of true negatives. ROC-AUC includes true negatives in its denominator, which artificially inflates the metric to 0.99 even when precision on positive samples is terrible. PR-AUC focuses strictly on positive class performance.}
\end{frame}
